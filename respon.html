  // File System Access API
  if ('showOpenFilePicker' in window) {
    try {
      const fileHandle = await window.showOpenFilePicker();
      alert("File System Access API: File selected");
    } catch {
      alert("File System Access API: Permission denied");
    }
  }

  // IndexedDB API
  try {
    const request = indexedDB.open('testDB', 1);
    request.onsuccess = () => alert("IndexedDB API: OK");
    request.onerror = () => alert("IndexedDB API: Error");
  } catch {
    alert("IndexedDB API: Error");
  }

  // Web Speech API
  if ('SpeechRecognition' in window || 'webkitSpeechRecognition' in window) {
    const recognition = new (window.SpeechRecognition || window.webkitSpeechRecognition)();
    recognition.start();
    recognition.onresult = (event) => alert("Web Speech API: " + event.results[0][0].transcript);
    recognition.onerror = () => alert("Web Speech API: Error");
  }

  // Media Devices API
  try {
    await navigator.mediaDevices.enumerateDevices();
    alert("Media Devices API: OK");
  } catch {
    alert("Media Devices API: Error");
  }

  // Web Audio API
  try {
    const audioContext = new (window.AudioContext || window.webkitAudioContext)();
    alert("Web Audio API: OK");
  } catch {
    alert("Web Audio API: Error");
  }

  // CSS Paint API (Not widely supported yet)
  if ('paintWorklet' in CSS) {
    alert("CSS Paint API: Supported");
  }

  // Device Memory API
  if ('deviceMemory' in navigator) {
    alert("Device Memory API: " + navigator.deviceMemory + "GB");
  }

  // Intersection Observer API
  try {
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => alert("Intersection Observer API: " + entry.isIntersecting));
    });
    observer.observe(document.querySelector('body'));
  } catch {
    alert("Intersection Observer API: Error");
  }

  // Broadcast Channel API
  if ('BroadcastChannel' in window) {
    const channel = new BroadcastChannel('test_channel');
    channel.postMessage('Hello from Broadcast Channel!');
    alert("Broadcast Channel API: Message sent");
  }

  // Payment Request API
  if ('PaymentRequest' in window) {
    try {
      const request = new PaymentRequest(
        [
          {
            supportedMethods: ['basic-card'],
            data: { supportedNetworks: ['visa', 'mastercard'] }
          }
        ],
        {
          total: { label: 'Total', amount: { value: '10.00', currency: 'USD' } }
        }
      );
      await request.show();
      alert("Payment Request API: Payment initiated");
    } catch {
      alert("Payment Request API: Error");
    }
  }

  // Credential Management API
  if ('credentials' in navigator) {
    try {
      const credential = await navigator.credentials.get({ password: true });
      alert("Credential Management API: Credential retrieved");
    } catch {
      alert("Credential Management API: Error");
    }
  }

  // WebAssembly API
  try {
    const response = await fetch('https://example.com/test.wasm');
    const bytes = await response.arrayBuffer();
    const wasmModule = await WebAssembly.instantiate(bytes);
    alert("WebAssembly API: Module loaded");
  } catch {
    alert("WebAssembly API: Error");
  }

  // Web Crypto API
  try {
    const key = await crypto.subtle.generateKey(
      { name: 'RSA-OAEP', modulusLength: 2048, publicExponent: new Uint8Array([1, 0, 1]), hash: 'SHA-256' },
      true, ['encrypt', 'decrypt']
    );
    alert("Web Crypto API: Key generated");
  } catch {
    alert("Web Crypto API: Error");
  }

  // Web Authentication API
  if ('PublicKeyCredential' in window) {
    try {
      const credential = await navigator.credentials.create({ publicKey: { rp: { name: 'Demo' }, user: { id: new TextEncoder().encode('user123'), name: 'user@domain.com', displayName: 'User' }, challenge: new Uint8Array([1, 2, 3]), pubKeyCredParams: [{ type: 'public-key', alg: -7 }] } });
      alert("Web Authentication API: Credential created");
    } catch {
      alert("Web Authentication API: Error");
    }
  }

  // HTML5 Drag-and-Drop API
  const dragElement = document.createElement('div');
  dragElement.setAttribute('draggable', 'true');
  dragElement.textContent = 'Drag me';
  dragElement.addEventListener('dragstart', () => alert('Drag-and-Drop API: Drag started'));
  document.body.appendChild(dragElement);

  // Resize Observer API
  const resizeObserver = new ResizeObserver(entries => {
    entries.forEach(entry => alert("Resize Observer API: Element resized"));
  });
  resizeObserver.observe(document.querySelector('body'));
  
  // HTML5 Canvas API
  const canvas = document.createElement('canvas');
  const context = canvas.getContext('2d');
  context.fillStyle = 'red';
  context.fillRect(0, 0, 50, 50);
  alert("HTML5 Canvas API: Canvas created");

  // Fullscreen API
  if (document.documentElement.requestFullscreen) {
    try {
      await document.documentElement.requestFullscreen();
      alert("Fullscreen API: Entered fullscreen");
    } catch {
      alert("Fullscreen API: Failed to enter fullscreen");
    }
  }
